
pipeline {
    agent any

    options {
        timestamps()
        disableConcurrentBuilds()
    }

    parameters {
        choice(
            name: 'CONFIRM_DESTROY',
            choices: ['NO', 'YES'],
            description: 'Select YES to enable Terraform resource destruction.'
        )
    }

    environment {
        AWS_REGION       = 'us-east-1'
        TF_IN_AUTOMATION = 'true'
        TF_INPUT         = 'false'
    }

    stages {

        stage('Verify Repository') {
            steps {
                sh '''
                    set -eu
                    echo "Repository checked out by Jenkins"
                    echo "Branch: ${GIT_BRANCH:-unknown}"
                    pwd
                    ls -la

                    test -f backend.tf
                    test -f project.tf
                '''
            }
        }

        stage('Install Terraform and AWS CLI') {
            steps {
                sh '''
                    set -eu

                    if ! command -v aws >/dev/null 2>&1; then
                        sudo apt-get update
                        sudo apt-get install -y awscli
                    fi

                    if ! command -v terraform >/dev/null 2>&1; then
                        sudo apt-get update
                        sudo apt-get install -y \
                            gnupg software-properties-common wget

                        wget -O- \
                          https://apt.releases.hashicorp.com/gpg \
                          | sudo gpg --dearmor \
                            -o /usr/share/keyrings/hashicorp-archive-keyring.gpg

                        echo "deb [signed-by=/usr/share/keyrings/hashicorp-archive-keyring.gpg arch=$(dpkg --print-architecture)] https://apt.releases.hashicorp.com $(grep -oP '(?<=UBUNTU_CODENAME=).*' /etc/os-release || lsb_release -cs) main" \
                          | sudo tee /etc/apt/sources.list.d/hashicorp.list

                        sudo apt-get update
                        sudo apt-get install -y terraform
                    fi

                    aws --version
                    terraform version
                '''
            }
        }

        stage('Configure AWS Credentials') {
            steps {
                withCredentials([
                    string(
                        credentialsId: 'aws-access-key-id',
                        variable: 'AWS_ACCESS_KEY_ID'
                    ),
                    string(
                        credentialsId: 'aws-secret-access-key',
                        variable: 'AWS_SECRET_ACCESS_KEY'
                    )
                ]) {
                    sh '''
                        set +x
                        set -eu

                        export AWS_DEFAULT_REGION="$AWS_REGION"

                        echo "Verifying AWS identity..."
                        aws sts get-caller-identity \
                            --query Account \
                            --output text
                    '''
                }
            }
        }

        stage('Terraform Destroy Plan') {
            steps {
                withCredentials([
                    string(
                        credentialsId: 'aws-access-key-id',
                        variable: 'AWS_ACCESS_KEY_ID'
                    ),
                    string(
                        credentialsId: 'aws-secret-access-key',
                        variable: 'AWS_SECRET_ACCESS_KEY'
                    )
                ]) {
                    sh '''
                        set +x
                        set -eu

                        export AWS_DEFAULT_REGION="$AWS_REGION"

                        echo "Checking Terraform formatting..."
                        terraform fmt -check -recursive

                        echo "Initializing existing Terraform backend..."
                        terraform init -input=false

                        echo "Validating Terraform configuration..."
                        terraform validate

                        echo "Creating destroy plan..."
                        terraform plan \
                            -destroy \
                            -input=false \
                            -out=destroy.tfplan

                        echo "Review the resources listed below:"
                        terraform show -no-color destroy.tfplan
                    '''
                }
            }
        }

        stage('Confirm and Destroy') {
            when {
                expression {
                    params.CONFIRM_DESTROY == 'YES'
                }
            }

            steps {
                script {
                    input(
                        message: 'Review the Terraform destroy plan in the console. Do you want to permanently destroy the listed resources?',
                        ok: 'Destroy Resources'
                    )
                }

                withCredentials([
                    string(
                        credentialsId: 'aws-access-key-id',
                        variable: 'AWS_ACCESS_KEY_ID'
                    ),
                    string(
                        credentialsId: 'aws-secret-access-key',
                        variable: 'AWS_SECRET_ACCESS_KEY'
                    )
                ]) {
                    sh '''
                        set +x
                        set -eu

                        export AWS_DEFAULT_REGION="$AWS_REGION"

                        test -f destroy.tfplan

                        echo "Applying the reviewed destroy plan..."
                        terraform apply \
                            -input=false \
                            destroy.tfplan

                        echo "Terraform destroy completed."
                    '''
                }
            }
        }
    }

    post {
        success {
            echo 'Terraform destroy workflow completed. Check the build log to confirm whether destruction was approved or only planned.'
        }

        failure {
            echo 'Terraform workflow failed. Review the Jenkins console log.'
        }

        always {
            sh 'rm -f destroy.tfplan tfplan || true'
            cleanWs()
        }
    }
}
